# === Build personality ========================================================
# By default we disable tests. These can be enabled by passing
# -Dbuild_tests=true to meson.
option(
  'build_tests',
  type: 'boolean',
  value: false,
  description: 'Build and run unit tests',
)

# === Pool geometry =============================================================
# fpc owns these two knobs and forwards them to the pool-allocator subproject.
# The pool slot size is derived from `fpc_filter_max_order` (the largest
# user-visible struct is `struct fpc_fir`, whose tap arrays grow linearly with
# the order), so it is not a separate option.
option(
  'fpc_max_instances',
  type: 'integer',
  value: 8,
  description: 'Maximum number of instances per internal pool',
)

option(
  'fpc_filter_max_order',
  type: 'integer',
  value: 64,
  description: 'Maximum FIR filter order (number of taps). Drives the pool slot size.',
)
